   404→   404→function OrdersTab({ enabled, compact }: { enabled: boolean; compact?: boolean }) {
   405→   405→  const { lang, currency } = useStore();
   406→   406→  const t = useDict(lang);
   407→   407→  const { data, loading } = useAdminData<{ orders: AdminOrder[] }>("orders", enabled);
   408→   408→  const [localOrders, setLocalOrders] = useState<AdminOrder[] | null>(null);
   409→   409→
   410→   410→  if (loading || !data) return compact ? <p className="text-sm text-muted-foreground">{t.common.loading}</p> : <SkeletonGrid count={3} />;
   411→   411→  const orders = localOrders ?? (compact ? data.orders.slice(0, 5) : data.orders);
   412→   412→
   413→   413→  if (orders.length === 0) {
   414→   414→    return (
   415→   415→      <EmptyState
   416→   416→        icon={<ShoppingCart className="h-6 w-6" />}
   417→   417→        title={t.admin.noOrders}
   418→   418→      />
   419→   419→    );
   420→   420→  }
   421→   421→
   422→   422→  const statusLabel = (s: string) => {
   423→   423→    const map: Record<string, string> = {
   424→   424→      PENDING: t.account.statusPending,
   425→   425→      CONFIRMED: lang === "fr" ? "Confirmée" : "Confirmed",
   426→   426→      PAID: t.account.statusPaid,
   427→   427→      SHIPPED: t.account.statusShipped,
   428→   428→      DELIVERED: t.account.statusDelivered,
   429→   429→      CANCELLED: t.account.statusCancelled,
   430→   430→    };
   431→   431→    return map[s] || s;
   432→   432→  };
   433→   433→
   434→   434→  const handleStatusChange = async (orderId: string, newStatus: string) => {
