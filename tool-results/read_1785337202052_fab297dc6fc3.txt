   168→/* ============ Sous-onglets (regroupement) ============ */
   169→function SubTabs({
   170→  tabs,
   171→  panels,
   172→}: {
   173→  tabs: { key: string; label: string; icon: React.ReactNode }[];
   174→  panels: { key: string; content: React.ReactNode }[];
   175→}) {
   176→  const [active, setActive] = useState(tabs[0]?.key);
   177→  const panel = panels.find((p) => p.key === active);
   178→
   179→  return (
   180→    <div>
   181→      {/* Sous-onglets */}
   182→      <div className="flex gap-1 mb-6 border-b border-border">
   183→        {tabs.map((tb) => (
   184→          <button
   185→            key={tb.key}
   186→            onClick={() => setActive(tb.key)}
   187→            className={cn(
   188→              "inline-flex items-center gap-1.5 px-4 py-2 text-[11px] tracking-luxe-sm uppercase font-sans transition-colors border-b-2 -mb-px whitespace-nowrap",
   189→              active === tb.key
   190→                ? "border-accent text-accent"
   191→                : "border-transparent text-muted-foreground hover:text-foreground"
   192→            )}
   193→          >
   194→            {tb.icon}
   195→            {tb.label}
   196→          </button>
   197→        ))}
   198→      </div>
   199→      {/* Contenu */}
   200→      {panel?.content}
   201→    </div>
   202→  );
   203→}

   234→function DashboardTab({ enabled }: { enabled: boolean }) {
   235→  const { lang, currency } = useStore();
   236→  const t = useDict(lang);
   237→  const { data, loading } = useAdminData<{ stats: AdminStats }>("stats", enabled);
   238→  const [localStats, setLocalStats] = useState<AdminStats | null>(null);
   239→
   240→  // Recharge les stats quand le dashboard monte
   241→  useEffect(() => {
   242→    fetch("/api/admin/stats", { cache: "no-store" })
   243→      .then((r) => (r.ok ? r.json() : null))
   244→      .then((d) => { if (d?.stats) setLocalStats(d.stats); })
   245→      .catch(() => {});
   246→  }, []);
   247→
   248→  const s = localStats ?? data?.stats;
   249→
   250→  if (loading || !s) {
   251→    return <SkeletonGrid count={4} />;
   252→  }
   253→
   254→  return (
   255→    <div className="space-y-8">
   256→      {/* Toggle Paiement */}
   257→      <PaymentToggle />
   258→
   259→      {/* KPIs */}
   260→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   261→        <KpiCard
   262→          icon={<Calendar className="h-5 w-5" />}
   263→          label={t.admin.revenueToday}
   264→          value={formatPrice(s.revenueToday, lang, currency)}

   268→          icon={<TrendingUp className="h-5 w-5" />}
   269→          label={t.admin.revenueMonth}

   275→          label={t.admin.pendingOrders}

   394→/* ============ Commandes ============ */
   395→const ORDER_STATUS_STYLES: Record<string, string> = {
   396→  PENDING: "text-amber-600 bg-amber-50",
   397→  CONFIRMED: "text-blue-600 bg-blue-50",
   398→  PAID: "text-accent bg-accent/10",
   399→  SHIPPED: "text-blue-600 bg-blue-50",

   404→function OrdersTab({ enabled, compact }: { enabled: boolean; compact?: boolean }) {
   405→  const { lang, currency } = useStore();
   406→  const t = useDict(lang);
   407→  const { data, loading } = useAdminData<{ orders: AdminOrder[] }>("orders", enabled);
   408→  const [localOrders, setLocalOrders] = useState<AdminOrder[] | null>(null);
   409→
   410→  if (loading || !data) return compact ? <p className="text-sm text-muted-foreground">{t.common.loading}</p> : <SkeletonGrid count={3} />;
   411→  const orders = localOrders ?? (compact ? data.orders.slice(0, 5) : data.orders);
   412→
   413→  if (orders.length === 0) {
   414→    return (
   415→      <EmptyState
   416→        icon={<ShoppingCart className="h-6 w-6" />}
   417→        title={t.admin.noOrders}
   418→      />
   419→    );
   420→  }
   421→
   422→  const statusLabel = (s: string) => {
   423→    const map: Record<string, string> = {
   424→      PENDING: t.account.statusPending,
   425→      CONFIRMED: lang === "fr" ? "Confirmée" : "Confirmed",
   426→      PAID: t.account.statusPaid,
   427→      SHIPPED: t.account.statusShipped,
   428→      DELIVERED: t.account.statusDelivered,
   429→      CANCELLED: t.account.statusCancelled,
   430→    };
   431→    return map[s] || s;
   432→  };
   433→
   434→  const handleStatusChange = async (orderId: string, newStatus: string) => {
   435→    await fetch(`/api/admin/orders/${orderId}`, {
   436→      method: "PATCH",
   437→      headers: { "Content-Type": "application/json" },
   438→      body: JSON.stringify({ status: newStatus }),
   439→    });
   440→    // Met à jour localement
   441→    setLocalOrders((prev) => {
   442→      const base = prev ?? data.orders;
   443→      return base.map((o) => o.id === orderId ? { ...o, status: newStatus } : o);
   444→    });
   445→  };
   446→
   447→  return (
   448→    <div className="space-y-3">
   449→      {orders.map((o) => {
   450→        const date = new Date(o.createdAt).toLocaleDateString(
   451→          lang === "fr" ? "fr-FR" : "en-US",
   452→          { day: "2-digit", month: "short", year: "numeric" }
   453→        );
   454→        const itemCount = o.items.reduce((n, i) => n + i.qty, 0);
   455→        const isWhatsApp = o.payment?.provider === "WHATSAPP";
   456→        return (
   457→          <div
   458→            key={o.id}
   459→            className="flex flex-col sm:flex-row sm:items-center justify-between gap-3 p-4 border border-border rounded-2xl hover:border-accent/40 transition-colors"
   460→          >
   461→            <div className="flex items-center gap-3 min-w-0">
   462→              <span className="h-10 w-10 rounded-full bg-secondary flex items-center justify-center text-muted-foreground shrink-0">
   463→                <Package className="h-4 w-4" />
   464→              </span>
   465→              <div className="min-w-0">
   466→                <p className="font-sans font-medium text-sm truncate flex items-center gap-2">
   467→                  #{o.id.slice(-6).toUpperCase()} · {o.fullName}
   468→                  {isWhatsApp && (
   469→                    <span className="px-1.5 py-0.5 rounded-full bg-[#25D366]/10 text-[#25D366] text-[8px] tracking-luxe-sm uppercase">WhatsApp</span>
   470→                  )}
   471→                </p>
   472→                <p className="text-xs text-muted-foreground">
   473→                  {date} · {itemCount} {t.account.items}
   474→                </p>
   475→              </div>
   476→            </div>
   477→            <div className="flex items-center gap-3">
   478→              {compact ? (
   479→                <span
   480→                  className={cn(
   481→                    "px-2.5 py-1 rounded-full text-[10px] tracking-luxe-sm uppercase font-sans",
   482→                    ORDER_STATUS_STYLES[o.status] || ""
   483→                  )}
   484→                >
   485→                  {statusLabel(o.status)}
   486→                </span>
   487→              ) : (
   488→                <select
   489→                  value={o.status}
   490→                  onChange={(e) => handleStatusChange(o.id, e.target.value)}
   491→                  className={cn(
   492→                    "h-8 px-2 text-[10px] tracking-luxe-sm uppercase font-sans rounded-full border-0 cursor-pointer",
   493→                    ORDER_STATUS_STYLES[o.status] || "bg-secondary"
   494→                  )}
   495→                >
   496→                  <option value="PENDING">{statusLabel("PENDING")}</option>
   497→                  <option value="CONFIRMED">{statusLabel("CONFIRMED")}</option>
   498→                  <option value="PAID">{statusLabel("PAID")}</option>
   499→                  <option value="SHIPPED">{statusLabel("SHIPPED")}</option>
   500→                  <option value="DELIVERED">{statusLabel("DELIVERED")}</option>
   501→                  <option value="CANCELLED">{statusLabel("CANCELLED")}</option>
   502→                </select>
   503→              )}
   504→              <span className="font-serif text-base whitespace-nowrap">
   505→                {formatPrice(o.totalCents, lang, currency)}
   506→              </span>
   507→            </div>
   508→          </div>
   509→        );
   510→      })}
   511→    </div>
   512→  );
   513→}

   746→function UsersTab({ enabled }: { enabled: boolean }) {

   750→  const [localUsers, setLocalUsers] = useState<AdminUser[] | null>(null);

   752→
   753→  const users = localUsers ?? data?.users ?? [];
   754→
   755→  const reload = async () => {
   756→    try {
   757→      const res = await fetch("/api/admin/users", { cache: "no-store" });
   758→      const d = await res.json();
   759→      setLocalUsers(d.users || []);
   760→    } catch {
   761→      /* ignore */
   762→    }
   763→  };
   764→
   765→  const changeRole = async (id: string, role: string) => {
   766→    await fetch(`/api/admin/users/${id}`, {
   767→      method: "PATCH",
   768→      headers: { "Content-Type": "application/json" },
   769→      body: JSON.stringify({ role }),
   770→    });
   771→    reload();
   772→  };
   773→
   774→  const toggleBlock = async (u: AdminUser) => {
   775→    await fetch(`/api/admin/users/${u.id}`, {
   776→      method: "PATCH",
   777→      headers: { "Content-Type": "application/json" },
   778→      body: JSON.stringify({ blocked: !u.blocked }),
   779→    });
   780→    reload();
   781→  };
   782→
   783→  const remove = async (u: AdminUser) => {
   784→    if (!confirm(`Supprimer ${u.name} ?`)) return;
   785→    await fetch(`/api/admin/users/${u.id}`, { method: "DELETE" });
   786→    reload();
   787→  };
   788→
   789→  if (loading && !data) return <SkeletonGrid count={4} />;
   790→
   791→  return (
   792→    <div>
   793→      <div className="flex items-center justify-between mb-4">
   794→        <p className="text-sm text-muted-foreground">
   795→          {users.length} {users.length > 1 ? "utilisateurs" : "utilisateur"}
   796→        </p>
   797→        <button
   798→          onClick={() => {
   799→            setEditorOpen(true);
   800→          }}
   801→          className="inline-flex items-center gap-2 bg-foreground text-background px-4 py-2.5 text-[11px] tracking-luxe-sm uppercase font-sans rounded-full hover:bg-accent hover:text-accent-foreground transition-colors"
   802→        >
   803→          <Plus className="h-4 w-4" />
   804→          {t.admin.newUser}
   805→        </button>
   806→      </div>
   807→
   808→      {users.length === 0 ? (
   809→        <EmptyState icon={<Users className="h-6 w-6" />} title={t.admin.noUsers} />
   810→      ) : (
   811→        <div className="space-y-3">
   812→          {users.map((u) => (
   813→            <div
   814→              key={u.id}
   815→              className="flex flex-col sm:flex-row sm:items-center justify-between gap-3 p-4 border border-border rounded-2xl"
   816→            >
   817→              <div className="flex items-center gap-3 min-w-0">
   818→                <span
   819→                  className={cn(
   820→                    "h-11 w-11 rounded-full flex items-center justify-center font-serif text-lg shrink-0",
   821→                    u.blocked ? "bg-destructive/10 text-destructive" : "bg-accent/10 text-accent"
   822→                  )}
   823→                >
   824→                  {u.name.charAt(0).toUpperCase()}
   825→                </span>
   826→                <div className="min-w-0">
   827→                  <p className="font-sans font-medium truncate flex items-center gap-2">
   828→                    {u.name}
   829→                    {u.blocked && (
   830→                      <span className="px-2 py-0.5 rounded-full bg-destructive/10 text-destructive text-[9px] tracking-luxe-sm uppercase">
   831→                        {t.admin.blocked}
   832→                      </span>
   833→                    )}
   834→                  </p>
   835→                  <p className="text-xs text-muted-foreground truncate">
   836→                    {u.email} · {u.orderCount} {t.admin.customerOrders}
   837→                  </p>
   838→                </div>
   839→              </div>
   840→              <div className="flex items-center gap-2">
   841→                {/* Sélecteur de rôle */}
   842→                <select
   843→                  value={u.role}
   844→                  onChange={(e) => changeRole(u.id, e.target.value)}
   845→                  className={cn(
   846→                    "h-9 px-3 text-[10px] tracking-luxe-sm uppercase font-sans rounded-full border-0 cursor-pointer",
   847→                    ROLE_STYLES[u.role] || "bg-secondary"
   848→                  )}
   849→                >
   850→                  <option value="ADMIN">{t.auth.adminBadge}</option>
   851→                  <option value="MANAGER">{t.admin.roleManager}</option>
   852→                  <option value="CUSTOMER">{t.auth.customerBadge}</option>
   853→                </select>
   854→                {/* Bloquer/Débloquer */}
   855→                <button
   856→                  onClick={() => toggleBlock(u)}
   857→                  title={u.blocked ? t.admin.unblock : t.admin.block}
   858→                  className="h-9 w-9 rounded-full border border-border flex items-center justify-center text-muted-foreground hover:bg-secondary hover:text-foreground transition-colors"
   859→                >

   877→        <UserEditor
   878→          onClose={() => {
   879→            setEditorOpen(false);

   891→function UserEditor({
   892→  onClose,
   893→  onSaved,
   894→}: {
   895→  onClose: () => void;
   896→  onSaved: () => void;
   897→}) {
   898→  const { lang } = useStore();
   899→  const t = useDict(lang);
   900→  const [form, setForm] = useState({
   901→    name: "",
   902→    email: "",
   903→    password: "",
   904→    phone: "",
   905→    role: "CUSTOMER",
   906→  });

   912→    if (!form.name || !form.email || !form.password || !form.role) {
   913→      return;
   914→    }
   915→    setSaving(true);
   916→    try {
   917→      const res = await fetch("/api/admin/users", {
   918→        method: "POST",
   919→        headers: { "Content-Type": "application/json" },
   920→        body: JSON.stringify(form),
   921→      });
   922→      if (!res.ok) {
   923→        return;
   924→      }
   925→      onSaved();
   926→    } finally {
   927→      setSaving(false);
   928→    }
   929→  };
   930→
   931→  return (
