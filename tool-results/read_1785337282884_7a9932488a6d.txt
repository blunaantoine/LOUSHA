    40→    40→  open,

    42→    42→  onSaved,

    51→    51→

    56→    56→        name: product.name,

    66→    66→        badge: product.badge,

    87→    87→        return;
    88→    88→      }

    90→    90→      toast.success(lang === "fr" ? "Image téléversée" : "Image uploaded");
    91→    91→
    92→    92→      // Auto-génération IA après upload (silencieux si indisponible)

    95→    95→      toast.error("Upload impossible");

    98→    98→    }

   100→   100→

   103→   103→  const handleAIGenerate = async (imageUrl?: string) => {

   105→   105→    if (!url) {
   106→   106→      return;
   107→   107→    }

   109→   109→    try {
   110→   110→      const res = await fetch("/api/admin/ai-generate", {
   111→   111→        method: "POST",
   112→   112→        headers: { "Content-Type": "application/json" },
   113→   113→        body: JSON.stringify({ imageUrl: url }),
   114→   114→      });
   115→   115→      const data = await res.json();
   116→   116→      if (!res.ok) {
   117→   117→        const errMsg = data.error || (lang === "fr" ? "IA indisponible" : "AI unavailable");
   118→   118→        console.error("[ai-generate] API error:", errMsg);
   119→   119→        toast(errMsg);
   120→   120→        return;
   121→   121→      }
   122→   122→      const g = data.generated;
   123→   123→      if (!form.name && g.name) update("name", g.name);
   124→   124→      if (!form.nameEn && g.nameEn) update("nameEn", g.nameEn);
   125→   125→      if (!form.slug && g.slug) update("slug", g.slug);
   126→   126→      if (!form.material && g.material) update("material", g.material);
   127→   127→      if (!form.craftingTime && g.craftingTime) update("craftingTime", g.craftingTime);
   128→   128→      toast.success(lang === "fr" ? "✨ Contenu généré par IA" : "✨ AI content generated");
   129→   129→    } catch (err) {
   130→   130→      console.error("[ai-generate] Network/client error:", err);
   131→   131→      toast(lang === "fr" ? "Erreur réseau — IA indisponible" : "Network error — AI unavailable");
   132→   132→    } finally {
   133→   133→      setAiLoading(false);
   134→   134→    }
   135→   135→  };
   136→   136→
   137→   137→  const handleSave = async () => {
   138→   138→    if (!form.name || !form.slug || !form.priceCents || !form.categorySlug || !form.image) {
   139→   139→      toast.error("Remplis les champs requis (slug, nom, prix, catégorie, image).");
   140→   140→      return;
   141→   141→    }
   142→   142→    setSaving(true);
   143→   143→    try {
   144→   144→      const url = product
   145→   145→        ? `/api/admin/products/${product.id}`
   146→   146→        : "/api/admin/products";
   147→   147→      const method = product ? "PATCH" : "POST";
   148→   148→      const res = await fetch(url, {
   149→   149→        method,
   150→   150→        headers: { "Content-Type": "application/json" },
   151→   151→        body: JSON.stringify(form),
   152→   152→      });
   153→   153→      const data = await res.json();
   154→   154→      if (!res.ok) {
   155→   155→        toast.error(data.error || "Échec");
   156→   156→        return;
   157→   157→      }
   158→   158→      toast.success(product ? "Produit modifié" : "Produit créé");
   159→   159→      onSaved();
   160→   160→      onClose();
   161→   161→    } catch {
   162→   162→      toast.error("Erreur réseau");
   163→   163→    } finally {
   164→   164→      setSaving(false);
   165→   165→    }
   166→   166→  };
   167→   167→
   168→   168→  const handleDelete = async () => {
   169→   169→    if (!product) return;
   170→   170→    if (!confirm("Supprimer ce produit ?")) return;
   171→   171→    setSaving(true);

   173→   173→      const res = await fetch(`/api/admin/products/${product.id}`, {
   174→   174→        method: "DELETE",
   175→   175→      });
   176→   176→      if (!res.ok) {
   177→   177→        toast.error("Suppression échouée");
   178→   178→        return;
   179→   179→      }
   180→   180→      toast.success("Produit supprimé");
   181→   181→      onSaved();
   182→   182→      onClose();
   183→   183→    } finally {
   184→   184→      setSaving(false);
   185→   185→    }

   192→   192→          {product ? "Modifier le produit" : "Nouveau produit"}

   196→   196→            {product ? t.admin.editProduct : t.admin.newProduct}

   199→   199→            <X className="h-5 w-5" />
   200→   200→          </button>
   201→   201→        </div>
   202→   202→
   203→   203→        <div className="p-6 space-y-5">
   204→   204→          {/* Image */}
   205→   205→          <div>
   206→   206→            <Label>{t.admin.productImage}</Label>
   207→   207→            <div className="flex items-center gap-4 mt-1.5">
   208→   208→              <div className="h-20 w-20 rounded-xl overflow-hidden bg-secondary shrink-0">
   209→   209→                {form.image ? (
   210→   210→                  <img src={form.image} alt="" className="h-full w-full object-cover" />
   211→   211→                ) : null}
   212→   212→              </div>
   213→   213→              <label className="inline-flex items-center gap-2 border border-border px-4 py-2.5 text-[11px] tracking-luxe-sm uppercase font-sans rounded-full cursor-pointer hover:bg-foreground hover:text-background transition-colors">
   214→   214→                {uploading ? (
   215→   215→                  <Loader2 className="h-4 w-4 animate-spin" />
   216→   216→                ) : (
   217→   217→                  <Upload className="h-4 w-4" />
   218→   218→                )}
   219→   219→                {t.admin.uploadImage}
   220→   220→                <input
   221→   221→                  type="file"
   222→   222→                  accept="image/*"
   223→   223→                  className="hidden"
   224→   224→                  onChange={(e) => {
   225→   225→                    const f = e.target.files?.[0];
   226→   226→                    if (f) handleUpload(f);
   227→   227→                  }}
   228→   228→                />
   229→   229→              </label>
   230→   230→              {/* Bouton génération IA */}
   231→   231→              {form.image && (
   232→   232→                <button
   233→   233→                  onClick={() => handleAIGenerate()}
   234→   234→                  disabled={aiLoading}
   235→   235→                  className="inline-flex items-center gap-2 bg-accent text-accent-foreground px-4 py-2.5 text-[11px] tracking-luxe-sm uppercase font-sans rounded-full hover:bg-accent/90 transition-colors disabled:opacity-60"
   236→   236→                  title={lang === "fr" ? "Générer le contenu avec l'IA" : "Generate content with AI"}
   237→   237→                >
   238→   238→                  {aiLoading ? (
   239→   239→                    <Loader2 className="h-4 w-4 animate-spin" />
   240→   240→                  ) : (
   241→   241→                    <Sparkles className="h-4 w-4" />
   242→   242→                  )}
   243→   243→                  {lang === "fr" ? "Générer IA" : "AI Generate"}
   244→   244→                </button>
   245→   245→              )}
   246→   246→            </div>
   247→   247→            {aiLoading && (
   248→   248→              <p className="text-xs text-accent mt-2 animate-pulse">
   249→   249→                {lang === "fr" ? "✨ L'IA analyse l'image..." : "✨ AI analyzing image..."}
   250→   250→              </p>
   251→   251→            )}
   252→   252→          </div>
   253→   253→
   254→   254→          <div className="grid grid-cols-2 gap-4">
   255→   255→            <Input label={`${t.admin.colProduct} (FR)`} value={form.name} onChange={(v) => update("name", v)} required />
   256→   256→            <Input label={`${t.admin.colProduct} (EN)`} value={form.nameEn} onChange={(v) => update("nameEn", v)} required />
   257→   257→            <Input label="Slug" value={form.slug} onChange={(v) => update("slug", v.toLowerCase().replace(/\s+/g, "-"))} required />
   258→   258→            <Select
   259→   259→              label={t.admin.colCategory}
   260→   260→              value={form.categorySlug}
   261→   261→              onChange={(v) => update("categorySlug", v)}
   262→   262→              options={categories.map((c) => ({ value: c.slug, label: lang === "fr" ? c.name : c.nameEn }))}
   263→   263→              required
   264→   264→            />

   270→   270→              required

   273→   273→              label={t.admin.colStock}
   274→   274→              type="number"
   275→   275→              value={String(form.stock)}
   276→   276→              onChange={(v) => update("stock", Number(v))}
   277→   277→            />
   278→   278→            <Select
   279→   279→              label="Badge"
   280→   280→              value={form.badge}
   281→   281→              onChange={(v) => update("badge", v)}
   282→   282→              options={[
   283→   283→                { value: "none", label: "Aucun" },
   284→   284→                { value: "new", label: t.products.new },
   285→   285→                { value: "bestseller", label: t.products.bestseller },
   286→   286→              ]}
   287→   287→            />
   288→   288→            <div className="flex items-end gap-4 pb-2">
   289→   289→              <Toggle label="Featured" checked={form.featured} onChange={(v) => update("featured", v)} />
   290→   290→              <Toggle label={t.admin.inStock} checked={form.inStock} onChange={(v) => update("inStock", v)} />
   291→   291→            </div>
   292→   292→          </div>
   293→   293→
   294→   294→          <div className="grid grid-cols-2 gap-4">
   295→   295→            <Input label={t.product.material} value={form.material} onChange={(v) => update("material", v)} />
   296→   296→            <Input label={t.product.origin} value={form.origin} onChange={(v) => update("origin", v)} />
   297→   297→          </div>
   298→   298→
   299→   299→          {form.priceCents > 0 && (
   300→   300→            <p className="text-sm text-muted-foreground">
   301→   301→              Aperçu prix : {formatPrice(form.priceCents, lang, "XOF")}
   302→   302→            </p>
   303→   303→          )}
   304→   304→        </div>
   305→   305→
   306→   306→        {/* === Section Variantes (édition uniquement) === */}
   307→   307→        {product && (
   308→   308→          <VariantManager productId={product.id} />
   309→   309→        )}
   310→   310→
   311→   311→        <div className="sticky bottom-0 flex items-center justify-between gap-3 px-6 py-4 border-t border-border bg-background">
   312→   312→          {product ? (
   313→   313→            <button
   314→   314→              onClick={handleDelete}
   315→   315→              disabled={saving}
   316→   316→              className="text-destructive text-[11px] tracking-luxe-sm uppercase font-sans hover:underline"
   317→   317→            >
   318→   318→              {t.admin.delete}
   319→   319→            </button>
   320→   320→          ) : (
   321→   321→            <span />
   322→   322→          )}
   323→   323→          <div className="flex gap-3">
   324→   324→            <button
   325→   325→              onClick={onClose}
   326→   326→              className="px-5 py-2.5 text-[11px] tracking-luxe-sm uppercase font-sans border border-border rounded-full hover:bg-secondary transition-colors"
   327→   327→            >
   328→   328→              {t.admin.cancel}
   329→   329→            </button>
   330→   330→            <button
   331→   331→              onClick={handleSave}
   332→   332→              disabled={saving || uploading}
   333→   333→              className="inline-flex items-center gap-2 px-5 py-2.5 text-[11px] tracking-luxe-sm uppercase font-sans bg-foreground text-background rounded-full hover:bg-accent hover:text-accent-foreground transition-colors disabled:opacity-60"
   334→   334→            >
   335→   335→              {saving && <Loader2 className="h-3.5 w-3.5 animate-spin" />}
   336→   336→              {t.admin.save}
   337→   337→            </button>
   338→   338→          </div>
   339→   339→        </div>
   340→   340→      </DialogContent>
   341→   341→    </Dialog>
   342→   342→  );
   343→   343→}

   464→   464→function VariantManager({ productId }: { productId: string }) {

   467→   467→  const [variants, setVariants] = useState<Variant[]>([]);
   468→   468→  const [loading, setLoading] = useState(true);
   469→   469→  const [showForm, setShowForm] = useState(false);
   470→   470→  const [newVariant, setNewVariant] = useState({
   471→   471→    label: "",
   472→   472→    labelEn: "",
   473→   473→    value: "",
   474→   474→    color: "",
   475→   475→    priceCents: 0,
   476→   476→    stock: 0,

   567→   567→                <Trash2 className="h-3.5 w-3.5" />

   569→   569→            </div>

   575→   575→      {showForm && (

   577→   577→          <div className="grid grid-cols-2 gap-3">

   580→   580→                {lang === "fr" ? "Libellé (FR)" : "Label (FR)"} *
   581→   581→              </span>
   582→   582→              <input
   583→   583→                value={newVariant.label}
   584→   584→                onChange={(e) => setNewVariant((v) => ({ ...v, label: e.target.value }))}
   585→   585→                className="w-full h-10 bg-background border border-border px-3 text-sm rounded-lg focus:outline-none focus:border-accent"

   587→   587→              />
   588→   588→            </label>

   590→   590→              <span className="block text-[10px] tracking-luxe-sm uppercase text-muted-foreground mb-1">
   591→   591→                {lang === "fr" ? "Libellé (EN)" : "Label (EN)"}
   592→   592→              </span>
